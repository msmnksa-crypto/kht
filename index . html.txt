    <script>
        function signPledge() {
            const name = document.getElementById('userName').value.trim();
            const grade = document.getElementById('userGrade').value.trim();
            
            if(!name || !grade) {
                alert('الرجاء إدخال الاسم والصف لتوقيع الوثيقة.');
                return;
            }

            document.getElementById('outputName').innerText = `${name} - ${grade}`;
            document.getElementById('pledge-form').style.display = 'none';
            document.getElementById('pledge-result').style.display = 'block';

            if(typeof confetti === 'function') {
                confetti({
                    particleCount: 100,
                    spread: 70,
                    origin: { y: 0.6 }
                });
            }
        }

        function resetPledge() {
            document.getElementById('userName').value = '';
            document.getElementById('userGrade').value = '';
            document.getElementById('pledge-form').style.display = 'block';
            document.getElementById('pledge-result').style.display = 'none';
        }

        function addMessage() {
            const content = document.getElementById('msgContent').value.trim();
            if(!content) {
                alert('الرجاء كتابة رسالة قبل النشر.');
                return;
            }

            const wallContainer = document.getElementById('wall-container');
            const newItem = document.createElement('div');
            newItem.className = 'wall-item';
            newItem.innerHTML = `<h4>طالبة مبادرة</h4><p>${content}</p>`;
            
            wallContainer.prepend(newItem);
            document.getElementById('msgContent').value = '';
            alert('تم نشر رسالتكِ في جدارية الوطن بنجاح!');
        }
    </script>
